# **94. 二叉树的中序遍历**

给定一个二叉树的根节点 `root`，返回它的中序遍历。

## 递归代码
```cpp
class Solution {
public:
    void dfs(TreeNode *u, vector<int> &ans)
    {
        if (u == nullptr) return ;
        dfs(u->left, ans);
        ans.push_back(u->val);
        dfs(u->right, ans);
    }
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int> ans;
        dfs(root, ans);
        return ans;
    }
};
```

## 用栈
```cpp
class Solution {
public:
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int> ans;
        stack<TreeNode*> stk;
        while(root != nullptr || !stk.empty()) {
            while(root != nullptr) {
                stk.push(root);
                root = root->left;
            }
            root = stk.top();
            stk.pop();
            ans.push_back(root->val);
            root = root->right;
        }
        return ans;
    }
};
```
